🚀 DÉMARRAGE RAPIDE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Installation (5 minutes)
   ────────────────────────────────────────────────────────────────────────
   pip install boto3
   sudo apt-get install postgresql-client
   
   # Installer AWS CLI
   curl "https://awscli.amazonaws.com/awscli-exe-linux-x86_64.zip" -o "awscliv2.zip"
   unzip awscliv2.zip
   sudo ./aws/install
   
   # Configurer AWS
   aws configure

2. Configuration du bucket S3
   ────────────────────────────────────────────────────────────────────────
   bash scripts/setup_s3_bucket.sh

3. Test de la configuration
   ────────────────────────────────────────────────────────────────────────
   python scripts/test_backup_setup.py

4. Première sauvegarde
   ────────────────────────────────────────────────────────────────────────
   python scripts/backup_database.py

📋 COMMANDES ESSENTIELLES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Sauvegarde:
  python scripts/backup_database.py
  python scripts/backup_database.py --no-s3              # Local uniquement
  python scripts/backup_database.py --s3-bucket autre    # Bucket personnalisé

Lister:
  python scripts/backup_database.py --list               # Sauvegardes locales
  python scripts/restore_from_s3.py list                 # Sauvegardes S3
  python scripts/restore_from_s3.py list --prefix 2026/02/  # Filtrer par date

Restaurer:
  python scripts/restore_from_s3.py restore 2026/02/20/OPCP_backup_20260220_143000.sql
  python scripts/restore_database.py backups/OPCP_backup_20260220_143000.sql

Automatisation (cron):
  0 2 * * * cd /path/to/OPCP && python3 scripts/backup_database.py >> /var/log/OPCP_backup.log 2>&1
